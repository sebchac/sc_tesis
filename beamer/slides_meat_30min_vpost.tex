\section{Motivación}
\begin{frame}{Pregunta de investigación}
    \setbeamertemplate{itemize items}[circle]
    \begin{wideitemize}
        \item En mercados agrícolas, la exposición a temperaturas extremas (\textbf{estrés térmico}) puede deteriorar el desempeño de los cultivos, incrementando los costos marginales de producción.
        
        \item La literatura ha analizado este fenómeno como un choque agregado de productividad que genera adaptabilidad por parte los agentes.
        \begin{itemize}
            \item Métodos de forma reducida, datos de panel, \textit{long differences}
        \end{itemize}
        \item Pero el estrés térmico también puede tener efectos sobre la interacción estratégica de los productores en un mercado oligopólico.

        \item \textbf{\color{amaranth} ¿Cómo el estrés térmico reconfigura el equilibrio estratégico y el bienestar en el mercado internacional del café?}
    \end{wideitemize}
\end{frame}

\begin{frame}{En este trabajo}
    \setbeamertemplate{itemize items}[circle]
    \begin{wideitemize}
        \item Estimación estructural del mercado internacional del café, bajo un modelo de competencia \textit{à la Stackelberg}

        \item Simulación de un escenario contrafactual sin estrés térmico. Efectos en precios, cantidades y excedente social.

        \item Descomposición del impacto total: efecto en costos + efecto estratégico.
        \begin{itemize}
            \item \textbf{¿Quién gana y quién pierde bajo el estrés térmico?}
        \end{itemize}
    \end{wideitemize}
    \end{frame}

\begin{frame}{El mercado internacional del café}
    \label{frame:mercado_supuestos}
    \setbeamertemplate{itemize items}[circle]
    \begin{wideitemize}
        \item \textbf{Un bien homogéneo:}

\section{Modelo}

\begin{frame}{\textit{Stackelberg} con  $m>1$ líderes}
    \setbeamertemplate{itemize items}[circle]
    \begin{wideitemize}
        \item Jerarquía de productores:
        \begin{itemize}
            \item $m = 3$ líderes (Brasil, Colombia, Vietnam).
            \item $n-m$ seguidores.
        \end{itemize}
        \item Función de demanda inversa lineal:
        $$ P_{t}(Q_{t}) = a_{t} - b Q_{t}, \quad Q_{t} = \sum_{i=1}^{m} q_{i,t}^{l} + \sum_{j=m+1}^{n} q_{j,t}^{s} $$
        \item Costos marginales dependen del estrés térmico:
        $$ c_{i,t} =  \alpha S_{i,t} + \phi_{t} + \phi_{i}, \quad \alpha > 0 $$
    \end{wideitemize}
\end{frame}

\begin{frame}{El modelo. \textit{Stackelberg} con  $m>1$ líderes}
    \setbeamertemplate{itemize items}[circle]
    \begin{wideitemize}
        \item Problema de optimización de líderes y seguidores:
\begin{equation}    
    \text{máx}_{q_{i}^{s} \ge 0}: \pi^{s}(q_{i}^{s}) = (A-B(\bar{Q}^{l}+Q^{s}_{-i} + q^{s}_{i})- c_{i})\cdot q_{i}^{s}
\end{equation}
\begin{equation} 
    \text{máx}_{q_{j}^{l} \ge 0}: \pi^{l}(q_{j}^{l}) = (A-B(Q^{l}_{-j}+Q^{s}(q^{l}_{j} + Q^{l}_{-j}) + q^{l}_{j})- c_{j})\cdot q_{j}^{l}
\end{equation}    
        \item Condiciones de primer orden:
\begin{equation}
    \label{eq:c_seguidor}
    c_{i}^{s} = P + \frac{\partial P}{\partial q_{i}^{s}}\cdot q_{i}^{s}
\end{equation}
\begin{equation}
    \label{eq:c_lider}
    c_{j}^{l} = P + \left( \frac{\partial P}{\partial Q^{s}} \cdot \frac{\partial Q^{s}}{\partial q_{j}^{l}} + \frac{\partial P}{\partial Q^{l}} \cdot \frac{\partial Q^{l}}{\partial q_{j}^{l}} \right)\cdot q_{j}^{l}
\end{equation}
    \end{wideitemize}
\end{frame}
