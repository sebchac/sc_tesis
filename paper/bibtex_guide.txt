%% NOTAS IMPORTANTES
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%

%% MÚLTIPLES AUTORES:
%% Usar "and" para separar autores (NO comas entre autores)
%% Ejemplo: author = {Smith, John and Doe, Jane and García, María}

%% ORGANIZACIONES COMO AUTOR:
%% Usar dobles llaves para evitar que se trate como nombre de persona
%% Ejemplo: author = {{International Coffee Organization}}

%% CARACTERES ESPECIALES:
%% Con UTF-8 (recomendado): author = {Pérez, José}
%% Con escapes LaTeX: author = {P{\'e}rez, Jos{\'e}}

%% DOI vs URL:
%% - DOI es OPCIONAL pero muy recomendado (formato estándar, permanente)
%% - Si hay DOI, no es necesario incluir URL
%% - URL solo si no hay DOI o para recursos exclusivamente web
%% - apacite formatea automáticamente los DOI como enlaces

%% RANGOS DE PÁGINAS:
